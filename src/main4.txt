#include <Arduino.h>
#include <ESP32Servo.h>

Servo myServo;
const int servoPin = 13;

void setup() {
  Serial.begin(115200);
  myServo.attach(servoPin);
  myServo.write(0);
  Serial.println("Servo ready");
}

void loop() {
  Serial.println("Moving to 90");
  myServo.write(90);
  delay(1000);
  
  Serial.println("Moving to 0");
  myServo.write(0);
  delay(1000);
}
